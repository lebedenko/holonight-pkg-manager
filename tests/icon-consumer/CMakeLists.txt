cmake_minimum_required(VERSION 3.25)
project(package-icon-consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Concurrent)
find_package(HolonightImages REQUIRED)
find_package(AppStreamQt 1.0 REQUIRED)
find_library(PACKAGE_ICONS NAMES holonight_package_icons PATHS "${ICON_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(package-icon-probe main.cpp)
target_include_directories(package-icon-probe PRIVATE "${ICON_PREFIX}/include")
target_link_libraries(package-icon-probe PRIVATE "${PACKAGE_ICONS}" HolonightImages::Images AppStreamQt Qt6::Gui Qt6::Concurrent)
